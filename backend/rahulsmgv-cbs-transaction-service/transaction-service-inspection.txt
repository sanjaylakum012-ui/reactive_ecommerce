package com.rahulsmgv.cbs.transaction.application.service;

import com.rahulsmgv.cbs.transaction.application.dto.CreateTransactionRequest;
import com.rahulsmgv.cbs.transaction.application.dto.TransactionMapper;
import com.rahulsmgv.cbs.transaction.application.dto.TransactionResponse;
import com.rahulsmgv.cbs.transaction.application.port.AccountService;
import com.rahulsmgv.cbs.transaction.application.port.TransactionRepository;
import com.rahulsmgv.cbs.transaction.domain.enums.TransactionType;
import com.rahulsmgv.cbs.transaction.domain.model.Money;
import com.rahulsmgv.cbs.transaction.domain.model.Transaction;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.Objects;
import java.util.UUID;

@Service
@Transactional
public class TransactionApplicationService {

    private final TransactionRepository transactionRepository;
    private final AccountService accountService;

    public TransactionApplicationService(
            TransactionRepository transactionRepository,
            AccountService accountService
    ) {
        this.transactionRepository = Objects.requireNonNull(
                transactionRepository,
                "Transaction repository must not be null"
        );

        this.accountService = Objects.requireNonNull(
                accountService,
                "Account service must not be null"
        );
    }

    public TransactionResponse create(
            CreateTransactionRequest request
    ) {
        Objects.requireNonNull(
                request,
                "Create transaction request must not be null"
        );

        /*
         * Idempotency protection.
         *
         * If the same idempotency key was already processed,
         * return the existing transaction instead of creating
         * another transaction.
         */
        var existingTransaction =
                transactionRepository.findByIdempotencyKey(
                        request.idempotencyKey()
                );

        if (existingTransaction.isPresent()) {
            return TransactionMapper.toResponse(
                    existingTransaction.get()
            );
        }

        /*
         * Transaction reference must also be unique.
         */
        var existingReference =
                transactionRepository.findByTransactionReference(
                        request.transactionReference()
                );

        if (existingReference.isPresent()) {
            throw new IllegalStateException(
                    "Transaction reference already exists: "
                            + request.transactionReference()
            );
        }

        validateAccounts(request);

        /*
         * Validate source account for transactions
         * that debit an account.
         */
        if (requiresSourceAccount(request.transactionType())) {

            validateAccountExists(
                    request.sourceAccountId()
            );

            validateAccountActive(
                    request.sourceAccountId()
            );

            validateSufficientBalance(
                    request.sourceAccountId(),
                    request.amount()
            );
        }

        /*
         * Validate destination account for transactions
         * that credit an account.
         */
        if (requiresDestinationAccount(request.transactionType())) {

            validateAccountExists(
                    request.destinationAccountId()
            );

            validateAccountActive(
                    request.destinationAccountId()
            );
        }

        Money money = new Money(
                request.amount(),
                request.currency()
        );

        Transaction transaction = Transaction.create(
                request.transactionReference(),
                request.idempotencyKey(),
                request.transactionType(),
                request.sourceAccountId(),
                request.destinationAccountId(),
                money,
                request.description()
        );

        transactionRepository.save(transaction);

        try {

            transaction.startProcessing();

            transaction.complete();

            transactionRepository.save(transaction);

            return TransactionMapper.toResponse(transaction);

        } catch (RuntimeException exception) {

            transaction.fail(
                    exception.getMessage() != null
                            ? exception.getMessage()
                            : "Transaction processing failed"
            );

            transactionRepository.save(transaction);

            throw exception;
        }
    }

    @Transactional(readOnly = true)
    public TransactionResponse getById(
            UUID transactionId
    ) {
        Objects.requireNonNull(
                transactionId,
                "Transaction ID must not be null"
        );

        Transaction transaction =
                transactionRepository.findById(transactionId)
                        .orElseThrow(() ->
                                new IllegalArgumentException(
                                        "Transaction not found: "
                                                + transactionId
                                )
                        );

        return TransactionMapper.toResponse(transaction);
    }

    @Transactional(readOnly = true)
    public TransactionResponse getByIdempotencyKey(
            String idempotencyKey
    ) {
        if (idempotencyKey == null || idempotencyKey.isBlank()) {
            throw new IllegalArgumentException(
                    "Idempotency key must not be blank"
            );
        }

        Transaction transaction =
                transactionRepository.findByIdempotencyKey(
                                idempotencyKey
                        )
                        .orElseThrow(() ->
                                new IllegalArgumentException(
                                        "Transaction not found for idempotency key: "
                                                + idempotencyKey
                                )
                        );

        return TransactionMapper.toResponse(transaction);
    }

    @Transactional(readOnly = true)
    public TransactionResponse getByReference(
            String transactionReference
    ) {
        if (transactionReference == null
                || transactionReference.isBlank()) {

            throw new IllegalArgumentException(
                    "Transaction reference must not be blank"
            );
        }

        Transaction transaction =
                transactionRepository.findByTransactionReference(
                                transactionReference
                        )
                        .orElseThrow(() ->
                                new IllegalArgumentException(
                                        "Transaction not found for reference: "
                                                + transactionReference
                                )
                        );

        return TransactionMapper.toResponse(transaction);
    }

    private void validateAccounts(
            CreateTransactionRequest request
    ) {
        TransactionType type = request.transactionType();

        switch (type) {

            case DEBIT -> {
                if (request.sourceAccountId() == null) {
                    throw new IllegalArgumentException(
                            "Source account is required for DEBIT"
                    );
                }

                if (request.destinationAccountId() != null) {
                    throw new IllegalArgumentException(
                            "Destination account must not be provided for DEBIT"
                    );
                }
            }

            case CREDIT -> {
                if (request.destinationAccountId() == null) {
                    throw new IllegalArgumentException(
                            "Destination account is required for CREDIT"
                    );
                }

                if (request.sourceAccountId() != null) {
                    throw new IllegalArgumentException(
                            "Source account must not be provided for CREDIT"
                    );
                }
            }

            case TRANSFER -> {
                if (request.sourceAccountId() == null) {
                    throw new IllegalArgumentException(
                            "Source account is required for TRANSFER"
                    );
                }

                if (request.destinationAccountId() == null) {
                    throw new IllegalArgumentException(
                            "Destination account is required for TRANSFER"
                    );
                }

                if (request.sourceAccountId()
                        .equals(request.destinationAccountId())) {

                    throw new IllegalArgumentException(
                            "Source and destination accounts must be different"
                    );
                }
            }
        }
    }

    private boolean requiresSourceAccount(
            TransactionType transactionType
    ) {
        return transactionType == TransactionType.DEBIT
                || transactionType == TransactionType.TRANSFER;
    }

    private boolean requiresDestinationAccount(
            TransactionType transactionType
    ) {
        return transactionType == TransactionType.CREDIT
                || transactionType == TransactionType.TRANSFER;
    }

    private void validateAccountExists(
            UUID accountId
    ) {
        if (!accountService.exists(accountId)) {
            throw new IllegalArgumentException(
                    "Account not found: " + accountId
            );
        }
    }

    private void validateAccountActive(
            UUID accountId
    ) {
        if (!accountService.isActive(accountId)) {
            throw new IllegalStateException(
                    "Account is not active: " + accountId
            );
        }
    }

    private void validateSufficientBalance(
            UUID accountId,
            java.math.BigDecimal amount
    ) {
        if (!accountService.hasSufficientBalance(
                accountId,
                amount
        )) {
            throw new IllegalStateException(
                    "Insufficient balance for account: "
                            + accountId
            );
        }
    }
}

import java.math.BigDecimal;

public record Money(
        BigDecimal amount,
        String currency
) {

    public Money {
        Objects.requireNonNull(amount, "Amount must not be null");
        Objects.requireNonNull(currency, "Currency must not be null");

        if (amount.signum() <= 0) {
            throw new IllegalArgumentException("Amount must be greater than zero");
        }

        if (currency.isBlank()) {
            throw new IllegalArgumentException("Currency must not be blank");
        }

        currency = currency.toUpperCase();
    }

package com.rahulsmgv.cbs.transaction.domain.model;

import com.rahulsmgv.cbs.transaction.domain.enums.TransactionDirection;
import com.rahulsmgv.cbs.transaction.domain.enums.TransactionStatus;
import com.rahulsmgv.cbs.transaction.domain.enums.TransactionType;

import java.time.Instant;
import java.util.Objects;
import java.util.UUID;

public class Transaction {

    private final UUID transactionId;
    private final String transactionReference;
    private final String idempotencyKey;

    private final TransactionType transactionType;

    private final UUID sourceAccountId;
    private final UUID destinationAccountId;

    private final Money amount;

    private TransactionStatus status;

    private String description;
    private String failureReason;

    private Instant createdAt;
    private Instant updatedAt;
    private Instant completedAt;

    private Transaction(
            UUID transactionId,
            String transactionReference,
            String idempotencyKey,
            TransactionType transactionType,
            UUID sourceAccountId,
            UUID destinationAccountId,
            Money amount,
            String description) {
        this.transactionId = Objects.requireNonNull(
                transactionId,
                "Transaction ID must not be null");

        this.transactionReference = requireText(
                transactionReference,
                "Transaction reference");

        this.idempotencyKey = requireText(
                idempotencyKey,
                "Idempotency key");

        this.transactionType = Objects.requireNonNull(
                transactionType,
                "Transaction type must not be null");

        this.amount = Objects.requireNonNull(
                amount,
                "Amount must not be null");

        validateAccounts(
                transactionType,
                sourceAccountId,
                destinationAccountId);

        this.sourceAccountId = sourceAccountId;
        this.destinationAccountId = destinationAccountId;

        this.description = normalize(description);

        this.status = TransactionStatus.INITIATED;

        this.createdAt = Instant.now();
        this.updatedAt = this.createdAt;
    }

    public static Transaction create(
            String transactionReference,
            String idempotencyKey,
            TransactionType transactionType,
            UUID sourceAccountId,
            UUID destinationAccountId,
            Money amount,
            String description) {
        return new Transaction(
                UUID.randomUUID(),
                transactionReference,
                idempotencyKey,
                transactionType,
                sourceAccountId,
                destinationAccountId,
                amount,
                description);
    }

    public static Transaction rehydrate(
            UUID transactionId,
            String transactionReference,
            String idempotencyKey,
            TransactionType transactionType,
            UUID sourceAccountId,
            UUID destinationAccountId,
            Money amount,
            String description,
            String failureReason,
            TransactionStatus status,
            Instant createdAt,
            Instant updatedAt,
            Instant completedAt) {
        Transaction transaction = new Transaction(
                transactionId,
                transactionReference,
                idempotencyKey,
                transactionType,
                sourceAccountId,
                destinationAccountId,
                amount,
                description);

        transaction.failureReason = failureReason;
        transaction.status = Objects.requireNonNull(
                status,
                "Transaction status must not be null");
        transaction.completedAt = completedAt;

        transaction.createdAt = Objects.requireNonNull(
                createdAt,
                "Created timestamp must not be null");
        transaction.updatedAt = Objects.requireNonNull(
                updatedAt,
                "Updated timestamp must not be null");

        return transaction;
    }

    public void startProcessing() {
        ensureStatus(TransactionStatus.INITIATED);

        this.status = TransactionStatus.PROCESSING;
        touch();
    }

    public void complete() {
        ensureStatus(TransactionStatus.PROCESSING);

        this.status = TransactionStatus.COMPLETED;
        this.completedAt = Instant.now();

        touch();
    }

    public void fail(String reason) {
        ensureStatus(
                TransactionStatus.INITIATED,
                TransactionStatus.PROCESSING);

        this.status = TransactionStatus.FAILED;
        this.failureReason = requireText(reason, "Failure reason");

        touch();
    }

    public void reverse() {
        ensureStatus(TransactionStatus.COMPLETED);

        this.status = TransactionStatus.REVERSED;
        touch();
    }

    public boolean isDebit() {
        return transactionType == TransactionType.DEBIT;
    }

    public boolean isCredit() {
        return transactionType == TransactionType.CREDIT;
    }

    public boolean isTransfer() {
        return transactionType == TransactionType.TRANSFER;
    }

    public boolean affectsAccount(
            UUID accountId,
            TransactionDirection direction) {
        Objects.requireNonNull(accountId, "Account ID must not be null");
        Objects.requireNonNull(direction, "Transaction direction must not be null");

        return switch (direction) {
            case DEBIT ->
                accountId.equals(sourceAccountId);

            case CREDIT ->
                accountId.equals(destinationAccountId);
        };
    }

    private void validateAccounts(
            TransactionType transactionType,
            UUID sourceAccountId,
            UUID destinationAccountId) {
        switch (transactionType) {

            case DEBIT -> {
                if (sourceAccountId == null) {
                    throw new IllegalArgumentException(
                            "Source account is required for DEBIT");
                }

                if (destinationAccountId != null) {
                    throw new IllegalArgumentException(
                            "Destination account must not be provided for DEBIT");
                }
            }

            case CREDIT -> {
                if (destinationAccountId == null) {
                    throw new IllegalArgumentException(
                            "Destination account is required for CREDIT");
                }

                if (sourceAccountId != null) {
                    throw new IllegalArgumentException(
                            "Source account must not be provided for CREDIT");
                }
            }

            case TRANSFER -> {
                if (sourceAccountId == null) {
                    throw new IllegalArgumentException(
                            "Source account is required for TRANSFER");
                }

                if (destinationAccountId == null) {
                    throw new IllegalArgumentException(
                            "Destination account is required for TRANSFER");
                }

                if (sourceAccountId.equals(destinationAccountId)) {
                    throw new IllegalArgumentException(
                            "Source and destination accounts must be different");
                }
            }
        }
    }

    private void ensureStatus(TransactionStatus... allowedStatuses) {
        for (TransactionStatus allowedStatus : allowedStatuses) {
            if (status == allowedStatus) {
                return;
            }
        }

        throw new IllegalStateException(
                "Invalid transaction state transition from "
                        + status);
    }

    private void touch() {
        this.updatedAt = Instant.now();
    }

    private static String requireText(
            String value,
            String field) {
        if (value == null || value.isBlank()) {
            throw new IllegalArgumentException(
                    field + " must not be blank");
        }

        return value.trim();
    }

    private static String normalize(String value) {
        if (value == null || value.isBlank()) {
            return null;
        }

        return value.trim();
    }

    public UUID getTransactionId() {
        return transactionId;
    }

    public String getTransactionReference() {
        return transactionReference;
    }

    public String getIdempotencyKey() {
        return idempotencyKey;
    }

    public TransactionType getTransactionType() {
        return transactionType;
    }

    public TransactionStatus getStatus() {
        return status;
    }

    public UUID getSourceAccountId() {
        return sourceAccountId;

        return destinationAccountId;
    }

        return amount;

        return description;
    }

    public String getFailureReason() {
        return failureReason;
    }

    public Instant getCreatedAt() {
        return createdAt;
    }

    public Instant getUpdatedAt() {
        return updatedAt;
    }

    public Instant getCompletedAt() {
        return completedAt;
    }
}

package com.rahulsmgv.cbs.transaction.infrastructure.persistence.adapter;

import com.rahulsmgv.cbs.transaction.application.port.TransactionRepository;
import com.rahulsmgv.cbs.transaction.domain.model.Money;
import com.rahulsmgv.cbs.transaction.domain.model.Transaction;
import com.rahulsmgv.cbs.transaction.infrastructure.persistence.entity.TransactionEntity;
import com.rahulsmgv.cbs.transaction.infrastructure.persistence.repository.SpringDataTransactionRepository;
import org.springframework.stereotype.Repository;

import java.util.Optional;
import java.util.UUID;

@Repository
public class TransactionRepositoryAdapter
        implements TransactionRepository {

    private final SpringDataTransactionRepository repository;

    public TransactionRepositoryAdapter(
            SpringDataTransactionRepository repository) {
        this.repository = repository;
    }

    @Override
    public Transaction save(Transaction transaction) {

        TransactionEntity entity = toEntity(transaction);

        TransactionEntity saved = repository.save(entity);

        return toDomain(saved);
    }

    @Override
    public Optional<Transaction> findById(
            UUID transactionId) {
        return repository.findById(transactionId)
                .map(this::toDomain);
    }

    @Override
    public Optional<Transaction> findByIdempotencyKey(
            String idempotencyKey) {
        return repository.findByIdempotencyKey(
                idempotencyKey)
                .map(this::toDomain);
    }

    @Override
    public Optional<Transaction> findByTransactionReference(
            String transactionReference) {
        return repository.findByTransactionReference(
                transactionReference)
                .map(this::toDomain);
    }

    private TransactionEntity toEntity(
            Transaction transaction) {
        return new TransactionEntity(
                transaction.getTransactionId(),
                transaction.getTransactionReference(),
                transaction.getIdempotencyKey(),
                transaction.getTransactionType(),
                transaction.getStatus(),
                transaction.getSourceAccountId(),
                transaction.getDestinationAccountId(),
                transaction.getAmount().amount(),
                transaction.getAmount().currency(),
                transaction.getDescription(),
                transaction.getFailureReason(),
                transaction.getCreatedAt(),
                transaction.getUpdatedAt(),
                transaction.getCompletedAt());
    }

    private Transaction toDomain(
            TransactionEntity entity) {
        return Transaction.rehydrate(
                entity.getTransactionId(),
                entity.getTransactionReference(),
                entity.getIdempotencyKey(),
                entity.getTransactionType(),
                entity.getSourceAccountId(),
                entity.getDestinationAccountId(),
                new Money(
                        entity.getAmount(),
                        entity.getCurrency()),
                entity.getDescription(),
                entity.getFailureReason(),
                entity.getStatus(),
                entity.getCreatedAt(),
                entity.getUpdatedAt(),
                entity.getCompletedAt());
    }
